-- Comandos SELECT  para selecionar datos especificos de una tabla--

-- select con where---

select ciudad, telefono from oficina where pais = 'españa'-- este comando selecciona la columna ciudad y telefono de la tabla oficina donde el pais sea igual a españa--
go  -- alt+39 para comillas simples

select ciudad, telefono from oficina where pais = 'EEUU' -- este comando selecciona la columna ciudad y telefono de la tabla oficina donde el pais sea igual a EEUU--
go


select nombre, apellido1, apellido2, email from empleado where codigo_jefe = 7 -- este comando selecciona las columnas nombre, apellidos y correo de la tabla empleado donde el código del jefe es igua a 7--
go


-- select con NULL---

select nombre, apellido1, apellido2, email from empleado where codigo_jefe is NULL -- este comando selecciona las columnas nombre, apellidos y correo de la tabla empleado donde el código del jefe es igua a nulo--
go  



-- select con excluir---

-- se utiliza el operador de negación (<> o !=) para excluir a los representantes de ventas.

go
